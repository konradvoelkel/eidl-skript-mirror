\begin{frame}
  \frametitle{ReLU}

  \begin{center}
    \Huge $\operatorname{ReLU}(y) = \max(0, y)$
  \end{center}

  \bigskip
  \begin{center}
    \normalsize Schwache Ableitung: $[y > 0]$, d.h.\ 1 für $y > 0$ und 0 sonst --\\
    bei $y = 0$ ist sie \emph{festgelegt}, nicht ausgerechnet.
  \end{center}

  \note{Dass man die Ableitung an einer Stelle einfach festlegen darf, ist genau der
    Inhalt einer Übungsaufgabe: schwache Ableitungen dürfen an einzelnen Punkten beliebig sein.
    Die Wahl 0 statt 1 spart Rechnungen.
    Name: \emph{Rectified} Linear Unit -- Gleichrichter, wie in der Elektrotechnik.}
\end{frame}

% ------------------------------------------- Rechenblock: vier Identitäten (2026-11-18)
\begin{frame}
  \frametitle{Nachrechnen I: die Identität}

  \begin{center}
    \LARGE $\operatorname{ReLU}(x) - \operatorname{ReLU}(-x)$
  \end{center}

  \medskip
  \begin{center}
    \Large
    \pause % Transition 2026-11-18: Fall für Fall mitrechnen lassen, nicht vorlesen
    $x > 0$:\quad $x - 0 = x$\\[6pt]
    \pause % Transition 2026-11-18
    $x < 0$:\quad $0 - (-x) = x$\\[6pt]
    \pause % Transition 2026-11-18
    $x = 0$:\quad $0 - 0 = 0$
  \end{center}

  \medskip
  \pause % Transition 2026-11-18: das Ergebnis erst, wenn alle drei Fälle stehen
  \begin{center}
    \LARGE $=\; x$
  \end{center}

  \note{Erste von vier Rechnungen, alle in derselben Form: Fallunterscheidung an der
    Knickstelle. Nach der zweiten kann der Hörsaal es allein -- das ist der Zweck der
    Wiederholung.
    \textbf{Die Pointe:} Zwei ReLU-Neuronen geben die Identität zurück. Ein ReLU-Netz kann
    seine Eingabe also \emph{durchreichen}, ohne Ausdrucksstärke zu verlieren. Das ist der
    Grund, warum man später Schichten überspringen darf (Residualverbindungen).
    Wer fragt, warum das nicht trivial ist: Die Identität ist linear, ReLU nicht. Die
    Linearität entsteht erst in der Kombination.}
\end{frame}

\begin{frame}
  \frametitle{Nachrechnen II: der Betrag}

  \begin{center}
    \LARGE $\operatorname{ReLU}(x) + \operatorname{ReLU}(-x)$
  \end{center}

  \medskip
  \begin{center}
    \Large
    \pause % Transition 2026-11-18
    $x > 0$:\quad $x + 0 = x$\\[6pt]
    \pause % Transition 2026-11-18
    $x < 0$:\quad $0 + (-x) = -x$\\[6pt]
    \pause % Transition 2026-11-18
    $x = 0$:\quad $0 + 0 = 0$
  \end{center}

  \medskip
  \pause % Transition 2026-11-18
  \begin{center}
    \LARGE $=\; |x|$

    \medskip
    \small An der Tafel war letzte Woche der Fall $x > 0$ untergegangen.
  \end{center}

  \note{\textbf{Richtigstellung.} In VL 6 ist diese Rechnung an der Tafel mit „also auch
    0`` geendet (Aufzeichnung ab 1:26); nachgesehen wurde nur der zweite Summand. Jetzt hat
    der Hörsaal es selbst gerechnet, also reicht ein Satz dazu.
    Dieselben zwei Bausteine wie eben, ein Vorzeichen anders -- einmal kommt etwas Lineares
    heraus, einmal nicht. Der Knick entsteht in der \emph{Kombination}, nicht im einzelnen
    ReLU.
    Addiert und subtrahiert man I und II, bekommt man die Umkehrung:
    $\operatorname{ReLU}(x) = (x + |x|)/2$. Das Paar $(x, |x|)$ trägt dieselbe Information
    wie die beiden Neuronen.
    $|x|$ ist das kleine Netz, mit dem wir später die Hälfte der Neuronen für $x^2$ sparen.}
\end{frame}

\begin{frame}
  \frametitle{Nachrechnen III: Skalieren bringt nichts}

  \begin{center}
    \LARGE $\operatorname{ReLU}(2x) - \operatorname{ReLU}(x)$
  \end{center}

  \medskip
  \begin{center}
    \Large
    \pause % Transition 2026-11-18
    $x > 0$:\quad $2x - x = x$\\[6pt]
    \pause % Transition 2026-11-18
    $x \le 0$:\quad $0 - 0 = 0$
  \end{center}

  \medskip
  \pause % Transition 2026-11-18: erst die Fälle, dann die Verallgemeinerung
  \begin{center}
    \LARGE $=\; \operatorname{ReLU}(x)$

    \medskip
    \normalsize allgemein $\operatorname{ReLU}(ax) = a\operatorname{ReLU}(x)$ für $a > 0$,\\[6pt]
    also auch $\operatorname{ReLU}(x) + \operatorname{ReLU}(2x) = 3\operatorname{ReLU}(x)$.
  \end{center}

  \note{Das ist die Rechnung, die am Ende von VL 6 für heute versprochen wurde -- hier
    einlösen und das auch sagen.
    \textbf{Merksatz:} Die Eingabe zu strecken verschiebt den Knick nicht und erzeugt keinen
    neuen. Skalieren ist umsonst. Damit ist die Frage offen, was denn dann einen zweiten
    Knick bringt -- Antwort auf der nächsten Folie.}
\end{frame}

\begin{frame}
  \frametitle{Nachrechnen IV: verschieben}

  \begin{center}
    \LARGE $\operatorname{ReLU}(x) + \operatorname{ReLU}(x-2)$
  \end{center}

  \medskip
  \begin{center}
    \Large
    \pause % Transition 2026-11-18
    $x < 0$:\quad $0 + 0 = 0$\\[6pt]
    \pause % Transition 2026-11-18
    $0 \le x < 2$:\quad $x + 0 = x$\\[6pt]
    \pause % Transition 2026-11-18
    $x \ge 2$:\quad $x + (x-2) = 2x - 2$
  \end{center}

  \medskip
  \pause % Transition 2026-11-18: die Gegenfrage treibt auf die nächste Folie
  \begin{center}
    \normalsize Drei Stücke, Steigungen $0$, $1$, $2$ -- und kein Name dafür.\\[6pt]
    Und mit $-\operatorname{ReLU}(x-2)$?
  \end{center}

  \note{Jetzt ist zum ersten Mal etwas entstanden, das sich nur \emph{stückweise}
    hinschreiben lässt. Genau so bauen wir gleich $x^2$.
    Wo sitzt der zweite Knick? Dort, wo das Argument null wird, also bei $x = 2$. Das ist
    die Beobachtung, aus der die Regel wird.
    Die Gegenfrage am Schluss nicht selbst beantworten -- die nächste Folie zeigt beides.}
\end{frame}

  \frametitle{Einmal steiler, einmal wieder flach}

  \begin{center}
    \includegraphics[width=0.81\textwidth]{vl07-knick-regel.png}
  \end{center}

  \pause % Transition 2026-11-18: erst die Bilder lesen lassen
  \begin{center}
    \small Ein negatives Gewicht nimmt Steigung wieder weg.

\begin{frame}
  \frametitle{Die Regel}

  \begin{center}
    \Large
    $w \cdot \operatorname{ReLU}(x - k)$\\[12pt]
    fügt an der Stelle $k$ einen Steigungssprung der Höhe $w$ hinzu\\[8pt]
    -- und sonst nichts.
  \end{center}

  \medskip
  \pause % Transition 2026-11-18
  \begin{center}
    \normalsize Skalieren trägt nichts bei (III), verschieben trägt alles (IV).\\[8pt]
    \textbf{Jeder Knick kostet ein Neuron.}
  \end{center}

  \note{Der Satz, der den Rest der Sitzung trägt. Ab hier ist $x^2$ nur noch Buchhaltung
    über Steigungssprünge, und bei der universellen Approximation kommt dieselbe Idee mit
    glatten Bausteinen wieder: zwei Sigmoide mit Ausgabegewichten $+h$ und $-h$.
    Rückwärts gelesen beantwortet die Regel auch, warum ein einzelnes ReLU-Neuron so wenig
    kann -- es hat genau einen Knick zu vergeben.}
\end{frame}

\begin{frame}
  \frametitle{Warum ReLU passt}

  \begin{columns}[c]
    \begin{column}{0.46\textwidth}
      \Large
      Linear + ReLU\\[8pt]
      $=$ Projektion auf einen Halbraum\\[14pt]
      \normalsize Summe von Knicken\\ $=$ stückweise linear
    \end{column}
    \begin{column}{0.5\textwidth}
      \includegraphics[width=\textwidth]{vl06-relu-sum.png}
    \end{column}
  \end{columns}

  \note{Ein ReLU-Neuron $\max(0, \langle w,x\rangle + b)$ ist null auf der einen Seite der
    Hyperebene und steigt linear auf der anderen. Die Knickstelle \emph{ist} die
    Hyperebene.
    Summiert man solche Knickfunktionen mit Gewichten, bekommt man stückweise lineare
    Funktionen -- und damit lässt sich jede stetige Funktion approximieren. Das bauen
    wir gleich.}
\end{frame}

\begin{frame}
  \frametitle{Problem: sterbende Neuronen}

  \begin{center}
    \Large
    Eingabe immer negativ $\;\Rightarrow\;$ Ausgabe immer 0\\[10pt]
    $\Rightarrow$ Ableitung 0 $\;\Rightarrow\;$ $\delta = 0$\\[10pt]
    $\Rightarrow$ die Gewichte ändern sich nie wieder.
  \end{center}

  \note{Das Neuron ist tot. Mit dem Bild aus VL 6: Das Fehlersignal kommt dort nicht mehr
    an, weil es mit null multipliziert wird.
    Bild vom flachen Land: Wer im flachen Gelände steht, bekommt vom Gradienten keine
    Richtung mehr.}
\end{frame}

\begin{frame}
  \frametitle{Zwei Auswege}

  \begin{columns}[c]
    \begin{column}{0.46\textwidth}
      \Large
      Leaky ReLU, ELU\\[8pt]
      \normalsize im Negativen nicht ganz flach\\[16pt]
      \Large Swish, GELU\\[8pt]
      \normalsize glatt: $y\,\sigma(y)$, \; $y\,\Phi(y)$
    \end{column}
    \begin{column}{0.5\textwidth}
      \includegraphics[width=\textwidth]{vl06-activations.png}
    \end{column}
  \end{columns}

  \note{GELU kam mit BERT und GPT-2 in die Transformer; die heutigen großen Sprachmodelle
    (Llama, Mistral, Qwen) nehmen Swish/SiLU im gated Feed-Forward-Block (SwiGLU).
    Am Plot diskutieren, was man eigentlich will: das Signal im Negativen \emph{behalten},
    ohne die Nichtlinearität zu verlieren. Die kleine Delle unter der Achse ist gewollt.
    Swish sieht aus wie das Nike-Logo -- das merkt sich der Hörsaal.}
\end{frame}

% ------------------------------------------------------------------ Boolesche Netze
\begin{frame}
  \frametitle{Von Hand bauen: NOT}

  \begin{center}
    \Huge $n(x) = 1 - x$
  \end{center}

  \bigskip
  \begin{center}
    \normalsize Ein Neuron: Gewicht $-1$, Bias $+1$, Aktivierung $\operatorname{id}$.
  \end{center}

  \note{Interpretation festlegen: $x \le 0$ heißt FALSE, $x \ge 1$ heißt TRUE.
    Wichtig und oft übersehen: Ein Netz ist auf \emph{ganz} $\mathbb{R}$ definiert, nicht
    nur auf den Booleschen Werten. Wir müssen also interpolieren (was ist $n(0{,}5)$?)
    \emph{und} extrapolieren (was ist $n(-1)$?).}
\end{frame}

\begin{frame}
  \frametitle{OR}

  \begin{center}
    \Large
    Ansatz $x + y$? \quad Nein: $f(1,1) = 2$.\\[12pt]
    \normalsize Ausweg: die Interpretation ändern --\\
    $x \le 0$ heißt FALSE, $x > 0$ heißt TRUE.\\[12pt]
    \Large Jetzt geht $[x + y > 0]$, und $x + y$ selbst auch. \quad Besser noch:\\[10pt]
    \Huge $\operatorname{ReLU}(x) + \operatorname{ReLU}(y)$
  \end{center}

  \note{Den ersten Ansatz bewusst scheitern lassen -- der Hörsaal findet den Fehler
    sofort, und das ist der beste Einstieg.
    \textbf{Dann aber den Interpretationswechsel ausdrücklich machen}, sonst widerspricht
    sich die Folie: Mit „$x \ge 1$ heißt TRUE`` von der NOT-Folie wäre auch
    $\operatorname{ReLU}(1) + \operatorname{ReLU}(1) = 2$ keine saubere 1. Erst „alles über
    null heißt TRUE`` macht die 2 unschädlich -- und damit den ersten Ansatz wieder
    brauchbar. So steht es auch im Skript.
    ReLU schneidet Negatives sauber ab, deshalb die letzte Variante.
    Frage: Was macht diese Funktion bei $(-1, 2)$? Das ist Extrapolation, und wir
    entscheiden sie mit der Wahl der Bausteine.}
\end{frame}

\begin{frame}
  \frametitle{OR und NOR auf ganz $\mathbb{R}^2$}

  \begin{center}
    \includegraphics[width=0.66\textwidth]{vl07-or-nor-heatmap.png}
  \end{center}

  \note{Die vier Booleschen Ecken sind markiert, die schwarze Linie ist die
    Nullstellenmenge.
    NOR ist NOT OR, also einfach hintereinandergesetzt:
    $g(x,y) = 1 - (\operatorname{ReLU}(x) + \operatorname{ReLU}(y))$ -- erste Schicht

    die Seiten. Das ist die Formel, die 2025 gezündet hat.}
\end{frame}

\begin{frame}
  \frametitle{Sechs Knicke -- welche Gewichte?}

  \begin{center}
    \includegraphics[width=0.50\textwidth]{vl07-square-interpolation.png}
  \end{center}

  \pause % Transition 2026-11-18: erst das Bild lesen lassen, dann die Aufgabe
  \begin{center}
    \small \textbf{Fünf Minuten:} Steigungen ablesen, Gewichte bestimmen, $Q(0)$ zur Probe.
  \end{center}

  \note{\textbf{Stillarbeit, Zeit ansagen und einhalten} -- die einzige Stelle der Sitzung,
    an der der Hörsaal allein rechnet.
    Was herauskommen soll: Stützstellen $(k, k^2)$, auf $[k, k+1]$ hat die Sekante die
    Steigung $2k+1$, von Stück zu Stück also immer $+2$. Links von $-3$ ist die Funktion
    konstant $9$, der erste Sprung ist $-5$. Mit der Regel von vorhin ist damit jedes
    Gewicht bestimmt.
    Für die Schnellen: Was passiert rechts von $3$? Die Steigung bleibt bei $5$ -- das ist
    die Extrapolation, die im Take-Home wiederkommt.}
\end{frame}

\begin{frame}
  \frametitle{$x^2$ aus ReLUs}

  \begin{center}
    \Large
    $Q(x) = 9 - 5\operatorname{ReLU}(x+3) + \sum_{k=-2}^{2} 2\operatorname{ReLU}(x-k)$
  \end{center}

  \bigskip
  \begin{center}
    \normalsize Probe: $Q(0) = 9 - 15 + 2\cdot 3 = 0$, \quad $Q(3) = 9 - 30 + 2 \cdot 15 = 9$.
  \end{center}

  \note{\textbf{Gemeinsam entwickeln, nicht vorlesen.} Die Interpolation durch $(k, k^2)$
    hat auf $[k, k+1]$ die Steigung $2k+1$ -- an jeder Stützstelle wächst die Steigung
    um 2, und jeder Knick ist eine ReLU.
    Das ist ein Netz mit einer versteckten Schicht aus 6 ReLU-Neuronen und einer linearen
    Ausgabeschicht.}
\end{frame}

\begin{frame}
  \frametitle{Die Interpolation}

  \begin{center}
    \includegraphics[width=0.50\textwidth]{vl07-square-interpolation.png}
  \end{center}

  \begin{center}
    \small $\textstyle Q(x) = 9 - 5\operatorname{ReLU}(x+3) + \sum_{k=-2}^{2} 2\operatorname{ReLU}(x-k)$
  \end{center}

  \note{Symmetrie nutzen: $|x| = \operatorname{ReLU}(x) + \operatorname{ReLU}(-x)$ ist
    selbst ein kleines Netz, und dann reicht die Interpolation auf $[0,3]$ mit nur drei
    Neuronen -- dafür ist das Netz tiefer.
    \emph{Weniger Neuronen pro Schicht gegen mehr Schichten} -- diesen Tausch sehen wir
    noch öfter.}
\end{frame}

\begin{frame}
  \frametitle{Von $x^2$ zu $xy$}

  \begin{center}
    \Huge $xy = \dfrac{(x+y)^2 - (x-y)^2}{4}$
  \end{center}

  \bigskip
  \begin{center}
    \normalsize $x+y$ und $x-y$ sind lineare Schichten.
  \end{center}

  \note{Die Polarisationsidentität -- kurz nachrechnen lassen.
    Mit MUL und Addition bekommt man alle Polynome, und mit Weierstraß alle stetigen
    Funktionen auf beschränkten Bereichen. Damit ist die Frage „geht das?“ beantwortet.
    Nebenbei: PyTorch könnte $x^2$ auch exakt rechnen (Gates, etwa in LSTMs) -- aber man
    weiß vorher nicht, ob man gerade $x^2$ braucht.}
\end{frame}

% ------------------------------------------------------------------ Universalität
\begin{frame}
  \frametitle{Universelle Approximation}

  \begin{center}
    \Large
    Mit genug Neuronen und Schichten lässt sich\\
    \emph{jede stetige} Funktion auf einem beschränkten Bereich\\
    beliebig gut approximieren.
  \end{center}

  \bigskip
  \begin{center}
    \normalsize Die richtige Frage ist nie „geht das?“,\\ sondern „wie geht das effizient?“
  \end{center}

  \note{Beweisskizze in fünf Schritten, falls jemand will: messbare Funktionen durch
    einfache Funktionen approximieren, einfache durch Stufenfunktionen, und eine Stufe
    baut man aus zwei steilen Sigmoiden mit entgegengesetzten Ausgabegewichten.
    Steht im Skript; hier reicht das Bild auf der nächsten Folie.}
\end{frame}

\begin{frame}
  \frametitle{Präziser}

  \begin{center}
    \Large
    Pinkus / Kidger--Lyons: schmale, aber \emph{tiefe} Netze reichen\\
    {\small (Schichten der Breite $d + D + 2$)}\\[16pt]
    Yarotsky (2017): $c\log(1/\varepsilon)$ Schichten, $c\,\varepsilon^{-d/k}$ Gewichte
  \end{center}

  \smallskip
  \begin{center}
    \small $f \colon E \to \mathbb{R}^D$ auf kompaktem $E \subseteq \mathbb{R}^d$: \;
    $d$ Eingabe-, $D$ Ausgabedimension, \;
    $k$ Glattheit ($k$-fach stetig differenzierbar), \; $\varepsilon$ Genauigkeit
  \end{center}

  \medskip
  \begin{center}
    \normalsize Genauigkeit kostet \emph{Tiefe} nur logarithmisch,\\
    \emph{Gewichte} aber exponentiell in der Eingabedimension.
  \end{center}

  \note{\textbf{Die Botschaft von Yarotsky in einem Satz:} Tiefe ist billig, Gewichte sind
    teuer -- und Glattheit macht sie billiger.
    Im Einzelnen: Die Zahl der Gewichte wächst exponentiell in der Eingabedimension -- der Fluch
    der Dimensionalität. Aber: glattere Funktionen (größeres $k$) brauchen weniger, und
    die \emph{Tiefe} wächst nur logarithmisch in der Genauigkeit. Das ist das Argument für
    \emph{tiefe} Netze: Die Genauigkeit holt man sich über Schichten, nicht über Breite.\\[2pt]
    \textbf{Und die Grenze des Satzes, die man dazusagen sollte:} Er sagt, dass die Gewichte
    \emph{existieren} -- genau wie alles andere heute. Er sagt nicht, dass ein Optimierer sie
    findet, und er sagt nichts darüber, ob das Netz aus endlich vielen Datenpunkten
    generalisiert. Das ist der Abschnitt „Approximieren heißt nicht generalisieren`` im
    Skript und die Überleitung zum Rest des Semesters.
    Am Rande, wenn Zeit ist: Jedes ReLU-Netz lässt sich auch als Entscheidungsbaum
    schreiben (Aytekin 2022) -- der Baum ist allerdings exponentiell groß.}
\end{frame}

\begin{frame}
  \frametitle{Take-Home}

  \begin{center}
    \Large
    Die Gewichte \emph{existieren} -- wir haben sie hingeschrieben.\\[16pt]
    Ob der Optimierer sie findet, ist eine zweite Frage.\\[16pt]
    \normalsize Aber: Netze interpolieren gut und extrapolieren schlecht.
  \end{center}

  \note{Der letzte Satz ist die wichtigste Warnung des Kapitels: Außerhalb der Stützstellen
    wird die Approximation beliebig schlecht. So rechnet auch die GPU den Sinus -- per
    Lookup-Table mit Interpolation, und nur im vorgesehenen Bereich.}
\end{frame}
